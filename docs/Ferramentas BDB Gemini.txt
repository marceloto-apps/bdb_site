import React, { useState, useMemo } from 'react';
import { LineChart, Line, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer, ReferenceLine, Area, ComposedChart } from 'recharts';

const App = () => {
  // Estados para os parâmetros da distribuição
  const [baseMean, setBaseMean] = useState(0);
  const [stdDev, setStdDev] = useState(1);
  const [skewness, setSkewness] = useState(0);
  const [kurtosis, setKurtosis] = useState(3);

  // A Média Real desloca-se com a assimetria (efeito visual de ser puxada pela cauda)
  // Em distribuições assimétricas, a média é mais sensível a valores extremos
  const actualMean = useMemo(() => {
    // A média desloca-se na direção da assimetria (cauda)
    return baseMean + (skewness * stdDev * 0.8);
  }, [baseMean, skewness, stdDev]);

  // Cálculo da densidade de probabilidade (PDF)
  // Incluímos um fator de escala para a curtose para afetar a amplitude vertical
  const calculatePDF = (x, m, sd, skew, kurt) => {
    const z = (x - m) / sd;
    const phi = (1 / (Math.sqrt(2 * Math.PI))) * Math.exp(-0.5 * z * z);
    
    // Ajuste de Gram-Charlier (Aproximação para Skewness e Kurtosis)
    const excessKurt = kurt - 3;
    const adjustment = 1 + (skew / 6) * (Math.pow(z, 3) - 3 * z) + 
                       (excessKurt / 24) * (Math.pow(z, 4) - 6 * Math.pow(z, 2) + 3);
    
    // Fator de amplitude: Curtose alta concentra massa no centro (pico mais alto/amplitude)
    const amplitudeFactor = 1 + (excessKurt * 0.12);
    
    const result = (phi / sd) * adjustment * amplitudeFactor;
    return Math.max(0, result);
  };

  // Gerar dados para o gráfico
  const data = useMemo(() => {
    const points = [];
    // Range fixo para o eixo X para facilitar a visualização do deslocamento geográfico
    const minX = -12;
    const maxX = 12;
    const step = 0.15;

    for (let x = minX; x <= maxX; x += step) {
      const y = calculatePDF(x, actualMean, stdDev, skewness, kurtosis);
      points.push({
        x: parseFloat(x.toFixed(2)),
        y: y,
        // Áreas de Z-Score baseadas na Média Real
        z1: Math.abs((x - actualMean) / stdDev) <= 1 ? y : 0,
        z2: Math.abs((x - actualMean) / stdDev) <= 2 ? y : 0,
        z3: Math.abs((x - actualMean) / stdDev) <= 3 ? y : 0,
      });
    }
    return points;
  }, [actualMean, stdDev, skewness, kurtosis]);

  // Cálculo das medidas centrais (Posições relativas pedagógicas)
  const stats = useMemo(() => {
    // Numa distribuição com assimetria positiva: Moda < Mediana < Média
    // A Moda resiste no pico, a Mediana fica no meio, a Média segue a cauda.
    const mode = actualMean - (skewness * stdDev * 0.6);
    const median = actualMean - (skewness * stdDev * 0.25);
    return { mode, median };
  }, [actualMean, skewness, stdDev]);

  return (
    <div className="flex flex-col h-screen bg-slate-50 p-4 md:p-8 font-sans">
      <div className="max-w-6xl mx-auto w-full bg-white rounded-2xl shadow-2xl overflow-hidden flex flex-col md:flex-row border border-slate-200">
        
        {/* Painel de Controle Lateral */}
        <div className="w-full md:w-80 bg-slate-900 p-8 text-white space-y-8">
          <div>
            <h2 className="text-2xl font-bold text-blue-400 tracking-tight">Parâmetros</h2>
            <p className="text-slate-400 text-xs mt-1 uppercase">Ajuste e veja o movimento</p>
          </div>
          
          <div className="space-y-6">
            <section>
              <label className="flex justify-between text-sm font-medium mb-2">
                <span>Centro Base (μ₀)</span>
                <span className="text-blue-400 font-mono">{baseMean}</span>
              </label>
              <input 
                type="range" min="-4" max="4" step="0.1" value={baseMean} 
                onChange={(e) => setBaseMean(parseFloat(e.target.value))}
                className="w-full h-1.5 bg-slate-700 rounded-lg appearance-none cursor-pointer accent-blue-500"
              />
            </section>

            <section>
              <label className="flex justify-between text-sm font-medium mb-2">
                <span>Desvio Padrão (σ)</span>
                <span className="text-emerald-400 font-mono">{stdDev}</span>
              </label>
              <input 
                type="range" min="0.6" max="2.5" step="0.1" value={stdDev} 
                onChange={(e) => setStdDev(parseFloat(e.target.value))}
                className="w-full h-1.5 bg-slate-700 rounded-lg appearance-none cursor-pointer accent-emerald-500"
              />
            </section>

            <section>
              <label className="flex justify-between text-sm font-medium mb-2">
                <span>Assimetria (Skew)</span>
                <span className="text-amber-400 font-mono">{skewness}</span>
              </label>
              <input 
                type="range" min="-2" max="2" step="0.1" value={skewness} 
                onChange={(e) => setSkewness(parseFloat(e.target.value))}
                className="w-full h-1.5 bg-slate-700 rounded-lg appearance-none cursor-pointer accent-amber-500"
              />
              <div className="flex justify-between text-[10px] text-slate-500 mt-1 uppercase font-bold tracking-tighter">
                <span>Esquerda</span>
                <span>Simétrica</span>
                <span>Direita</span>
              </div>
            </section>

            <section>
              <label className="flex justify-between text-sm font-medium mb-2">
                <span>Curtose (Amplitude)</span>
                <span className="text-purple-400 font-mono">{kurtosis.toFixed(1)}</span>
              </label>
              <input 
                type="range" min="1.5" max="6" step="0.1" value={kurtosis} 
                onChange={(e) => setKurtosis(parseFloat(e.target.value))}
                className="w-full h-1.5 bg-slate-700 rounded-lg appearance-none cursor-pointer accent-purple-500"
              />
              <p className="text-[10px] text-slate-500 mt-1 uppercase">Muda a altura do pico</p>
            </section>
          </div>

          <div className="pt-6 border-t border-slate-800 space-y-4">
            <h3 className="text-xs font-bold text-slate-500 uppercase tracking-widest">Legenda</h3>
            <div className="space-y-3">
              <div className="flex items-center gap-3 text-sm">
                <div className="w-4 h-1 bg-rose-500 rounded-full"></div> 
                <span className="text-slate-300">Média (Desloca com Skew)</span>
              </div>
              <div className="flex items-center gap-3 text-sm">
                <div className="w-4 h-1 bg-green-500 border-t border-dashed"></div> 
                <span className="text-slate-300">Mediana (Intermediária)</span>
              </div>
              <div className="flex items-center gap-3 text-sm">
                <div className="w-4 h-1 bg-amber-500 border-t border-dotted"></div> 
                <span className="text-slate-300">Moda (Ponto Mais Alto)</span>
              </div>
            </div>
          </div>
        </div>

        {/* Área Principal do Gráfico */}
        <div className="flex-1 p-8 flex flex-col bg-white">
          <header className="mb-8 flex justify-between items-start">
            <div>
              <h1 className="text-3xl font-black text-slate-900 tracking-tight">Laboratório Estatístico</h1>
              <p className="text-slate-500 text-lg italic mt-1 font-serif">A dinâmica das medidas de tendência central.</p>
            </div>
            <div className="bg-slate-100 px-4 py-2 rounded-full text-slate-600 font-mono text-sm border border-slate-200">
              Amplitude Pico: {(calculatePDF(stats.mode, actualMean, stdDev, skewness, kurtosis)).toFixed(2)}
            </div>
          </header>

          <div className="flex-1 min-h-[450px] relative">
            <ResponsiveContainer width="100%" height="100%">
              <ComposedChart data={data} margin={{ top: 20, right: 30, left: 0, bottom: 20 }}>
                <CartesianGrid strokeDasharray="3 3" vertical={false} stroke="#f1f5f9" />
                <XAxis 
                  dataKey="x" 
                  type="number" 
                  domain={[-10, 10]} 
                  label={{ value: 'Eixo X (Valores)', position: 'insideBottom', offset: -10 }} 
                />
                <YAxis domain={[0, 0.8]} hide />
                
                <Tooltip 
                  contentStyle={{ borderRadius: '16px', border: 'none', boxShadow: '0 20px 25px -5px rgba(0,0,0,0.1)' }}
                  labelFormatter={(v) => Valor: ${v}}
                  formatter={(v) => [v.toFixed(3), 'Densidade']}
                />
                
                {/* Áreas de Desvio Padrão (Z-Score) */}
                <Area type="monotone" dataKey="z3" stroke="none" fill="#f8fafc" isAnimationActive={false} />
                <Area type="monotone" dataKey="z2" stroke="none" fill="#f1f5f9" isAnimationActive={false} />
                <Area type="monotone" dataKey="z1" stroke="none" fill="#e2e8f0" isAnimationActive={false} />
                
                {/* Linha da Curva */}
                <Line type="monotone" dataKey="y" stroke="#0f172a" strokeWidth={4} dot={false} isAnimationActive={false} />
                
                {/* Marcadores de Tendência Central */}
                <ReferenceLine x={actualMean} stroke="#f43f5e" strokeWidth={3} label={{ position: 'top', value: 'Média', fill: '#f43f5e', fontSize: 13, fontWeight: 'bold' }} />
                <ReferenceLine x={stats.median} stroke="#22c55e" strokeDasharray="4 4" strokeWidth={2} label={{ position: 'top', value: 'Mediana', fill: '#22c55e', fontSize: 11 }} />
                <ReferenceLine x={stats.mode} stroke="#f59e0b" strokeDasharray="2 2" strokeWidth={2} label={{ position: 'top', value: 'Moda', fill: '#f59e0b', fontSize: 11 }} />

                {/* Marcadores de SD (Z-Scores) */}
                <ReferenceLine x={actualMean + stdDev} stroke="#94a3b8" strokeDasharray="3 3" label={{ value: '+1σ', position: 'bottom', fontSize: 10 }} />
                <ReferenceLine x={actualMean - stdDev} stroke="#94a3b8" strokeDasharray="3 3" label={{ value: '-1σ', position: 'bottom', fontSize: 10 }} />
              </ComposedChart>
            </ResponsiveContainer>
          </div>

          {/* Dicas Pedagógicas */}
          <div className="mt-8 grid grid-cols-1 md:grid-cols-2 gap-4">
            <div className="p-5 bg-rose-50 rounded-2xl border border-rose-100">

              </p>
            </div>
            <div className="p-5 bg-purple-50 rounded-2xl border border-purple-100">
              <h4 className="text-purple-700 font-bold text-sm uppercase tracking-wider mb-2">Controlo de Amplitude</h4>

              </p>
            </div>
          </div>
        </div>
      </div>
    </div>
  );
};

export default App;

import React, { useState, useCallback, useEffect } from 'react';
import { 
  LineChart, Line, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer, BarChart, Bar, Cell
} from 'recharts';
import { 
  Calculator, Zap, DollarSign, BarChart3, AlertTriangle, Activity, ShieldCheck, Clock
} from 'lucide-react';

/**
 * Big Data Bet - Versão v9 (Especial: Tempo de Amostra)
 * Análise de Risco Real e Eficiência Técnica com métricas de tempo.
 */

const App = () => {
  const [inputs, setInputs] = useState({
    banca: 1000,
    oddsMedia: 2.00,
    roiEsperado: 5.0,
    numBets: 1000,
    tempoMeses: 6,
    limiteDrawdown: 25,
    stakeEscolhida: 1.0, 
    simulacoesCount: 1000
  });

  const [entradasPorMes, setEntradasPorMes] = useState(166.7);
  const [results, setResults] = useState(null);
  const [isCalculating, setIsCalculating] = useState(false);

  // Atualiza o cálculo de entradas por mês sempre que o volume ou o tempo mudam
  useEffect(() => {
    const calc = inputs.numBets / inputs.tempoMeses;
    setEntradasPorMes(Number(calc.toFixed(1)));
  }, [inputs.numBets, inputs.tempoMeses]);

  const cumulativeNormal = (z) => {
    const t = 1 / (1 + 0.2316419 * Math.abs(z));
    const d = 0.3989423 * Math.exp(-z * z / 2);
    const prob = d * t * (0.3193815 + t * (-0.3565638 + t * (1.7814779 + t * (-1.821256 + t * 1.330274))));
    return z > 0 ? 1 - prob : prob;
  };

  const calculateStatistics = useCallback(() => {
    setIsCalculating(true);
    
    setTimeout(() => {
      const { banca, oddsMedia, roiEsperado, numBets, limiteDrawdown, stakeEscolhida, simulacoesCount } = inputs;
      
      const p = (roiEsperado / 100 + 1) / oddsMedia; 
      const q = 1 - p;
      const b = oddsMedia - 1;
      const roiDecimal = roiEsperado / 100;
      const s = stakeEscolhida / 100;

      // 1. P-Value
      const pNula = 1 / oddsMedia;
      const zScore = (p * numBets - pNula * numBets) / Math.sqrt(numBets * pNula * (1 - pNula));
      const pValue = 1 - cumulativeNormal(zScore);

      // 2. Volume Validador
      const variancePerUnit = (p * Math.pow(b, 2)) + (q * Math.pow(-1, 2)) - Math.pow(roiDecimal, 2);
      const stdDevPerUnit = Math.sqrt(variancePerUnit);
      const precisionNeeded = Math.max(0.005, roiDecimal * 0.2); 
      const volumeCalculado = Math.ceil(Math.pow((1.96 * stdDevPerUnit) / precisionNeeded, 2));

      // 3. Intervalos de Confiança
      const errorROI = stdDevPerUnit / Math.sqrt(numBets);
      const piorROI = (roiDecimal - 1.96 * errorROI) * 100;
      const melhorROI = (roiDecimal + 1.96 * errorROI) * 100;

      // 4. Monte Carlo
      let ddBuckets = Array(10).fill(0); 
      let failures = 0;
      let profits = 0;
      let sumMDD = 0;
      let worstDD = 0;
      let paths = [];
      const step = Math.max(1, Math.floor(numBets / 60));

      for (let simIdx = 0; simIdx < simulacoesCount; simIdx++) {
        let currentBanca = banca;
        let peak = banca;
        let localMDD = 0;
        let currentPath = [];
        
        for (let i = 0; i <= numBets; i++) {
          if (simIdx < 12 && (i % step === 0 || i === numBets)) {
             currentPath.push({ x: i, y: Math.round(currentBanca) });
          }
          const win = Math.random() < p;
          const betSize = currentBanca * s;
          currentBanca += win ? (betSize * b) : -betSize;
          
          if (currentBanca > peak) peak = currentBanca;
          const currentDD = peak > 0 ? ((peak - currentBanca) / peak) * 100 : 100;
          if (currentDD > localMDD) localMDD = currentDD;
          
          if (currentBanca <= 1) { 
            localMDD = 100; 
            currentBanca = 0; 
            break; 
          }
        }
        
        const bIdx = Math.min(Math.floor(localMDD / 10), 9);
        ddBuckets[bIdx]++;
        if (localMDD > limiteDrawdown) failures++;
        if (currentBanca > banca) profits++;
        sumMDD += localMDD;
        if (localMDD > worstDD) worstDD = localMDD;
        if (simIdx < 12) paths.push(currentPath);
      }

      const chartData = [];
      if (paths.length > 0) {
        for (let i = 0; i < paths[0].length; i++) {
          const entry = { bet: paths[0][i].x };
          paths.forEach((p, idx) => { if (p[i]) entry[s${idx}] = p[i].y; });
          chartData.push(entry);
        }
      }

      setResults({
        pValue,
        probLucro: (profits / simulacoesCount) * 100,
        survivalRate: ((simulacoesCount - failures) / simulacoesCount) * 100,
        volumeNecessario: volumeCalculado,
        piorROI,
        melhorROI,
        totalProfit: numBets * roiDecimal,
        avgMDD: sumMDD / simulacoesCount,
        worstDD,
        histData: ddBuckets.map((count, i) => ({
          range: ${i*10}%,
          percent: (count / simulacoesCount) * 100,
          danger: (i * 10) >= limiteDrawdown
        })),
        chartData
      });
      setIsCalculating(false);
    }, 400);
  }, [inputs]);

  return (
    <div className="min-h-screen bg-[#0d1117] text-slate-200 p-4 md:p-8 font-sans">
      <div className="max-w-7xl mx-auto space-y-6">
        
        {/* Header Profissional conforme Screenshot */}
        <div className="bg-[#161b22] border border-slate-800 p-6 rounded-2xl flex flex-col md:flex-row justify-between items-center gap-4 shadow-2xl">
          <div className="flex items-center gap-4">
            <div className="bg-yellow-400 p-2 rounded-lg">
              <Zap className="text-black fill-black" size={20} />
            </div>
            <div>
              <h1 className="text-2xl font-black text-white uppercase tracking-tight italic">BIG DATA BET: VALIDAÇÃO E RISCO</h1>
              <p className="text-blue-500 text-[10px] font-bold uppercase italic">Análise de Risco Real e Eficiência Técnica</p>
            </div>
          </div>
          <button 
            onClick={calculateStatistics}
            disabled={isCalculating}
            className="w-full md:w-auto bg-blue-600 hover:bg-blue-500 text-white font-black py-3 px-10 rounded-xl transition-all shadow-lg active:scale-95 disabled:opacity-50 uppercase text-sm tracking-widest"
          >
            {isCalculating ? "Processando..." : Simular ${inputs.simulacoesCount.toLocaleString()} Cenários}
          </button>
        </div>

        <div className="grid grid-cols-1 lg:grid-cols-12 gap-6">
          
          {/* PAINEL DE ENTRADA (Refletindo a Screenshot) */}
          <div className="lg:col-span-4 space-y-6">
            <div className="bg-[#161b22] border border-slate-800 p-6 rounded-2xl shadow-xl space-y-6">
              <div className="flex items-center gap-2 border-b border-slate-800 pb-3">
                <Calculator size={14} className="text-slate-500"/>
                <h2 className="text-[10px] font-black text-slate-500 uppercase tracking-widest">Parâmetros de Entrada</h2>
              </div>
              
              <div className="grid grid-cols-2 gap-4">
                <div className="space-y-1">
                  <label className="text-[9px] font-bold text-slate-400 uppercase ml-1">Banca Inicial</label>
                  <div className="bg-[#0d1117] p-3 rounded-lg border border-slate-800 focus-within:border-blue-500/50 transition-colors">
                    <input type="number" value={inputs.banca} onChange={e => setInputs({...inputs, banca: Number(e.target.value)})} className="bg-transparent w-full text-white font-black outline-none text-lg" />
                  </div>
                </div>
                <div className="space-y-1">
                  <label className="text-[9px] font-bold text-slate-400 uppercase ml-1">ROI Esperado %</label>
                  <div className="bg-[#0d1117] p-3 rounded-lg border border-slate-800 focus-within:border-emerald-500/50 transition-colors">
                    <input type="number" step="0.1" value={inputs.roiEsperado} onChange={e => setInputs({...inputs, roiEsperado: Number(e.target.value)})} className="bg-transparent w-full text-emerald-400 font-black outline-none text-lg" />
                  </div>
                </div>
              </div>

              <div className="grid grid-cols-2 gap-4">
                <div className="space-y-1">
                  <label className="text-[9px] font-bold text-slate-400 uppercase ml-1">Odd Média</label>
                  <div className="bg-[#0d1117] p-3 rounded-lg border border-slate-800">
                    <input type="number" step="0.01" value={inputs.oddsMedia} onChange={e => setInputs({...inputs, oddsMedia: Number(e.target.value)})} className="bg-transparent w-full text-white font-black outline-none text-lg" />
                  </div>
                </div>
                <div className="space-y-1">
                  <label className="text-[9px] font-bold text-slate-400 uppercase ml-1">Total de Bets</label>
                  <div className="bg-[#0d1117] p-3 rounded-lg border border-slate-800">
                    <input type="number" value={inputs.numBets} onChange={e => setInputs({...inputs, numBets: Number(e.target.value)})} className="bg-transparent w-full text-white font-black outline-none text-lg" />
                  </div>
                </div>
              </div>

              {/* Seção de Tempo - Replicada da Screenshot */}
              <div className="bg-[#0d1117] p-4 rounded-xl border border-slate-800 flex justify-between items-center">
                <div className="space-y-1">
                  <label className="text-[9px] font-black text-blue-400 uppercase flex items-center gap-1">
                    <Clock size={10} /> Tempo Amostra (Meses)
                  </label>
                  <input 
                    type="number" value={inputs.tempoMeses} 
                    onChange={e => setInputs({...inputs, tempoMeses: Math.max(1, Number(e.target.value))})} 
                    className="bg-transparent text-white font-black text-xl outline-none w-20" 
                  />
                </div>
                <div className="text-right">
                  <label className="text-[9px] font-black text-slate-500 uppercase">Entradas/Mês</label>
                  <div className="text-xl font-black text-white">{entradasPorMes}</div>

                  <label className="text-[9px] font-black text-blue-400 uppercase tracking-wider">Stake Atual Escolhida (%)</label>
